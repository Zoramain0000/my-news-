        /* --- Drawing Graphs --- */
        function drawGraph(ctx, data, startX, startY, color, bgColor) {
            const canvas = ctx.canvas;
            const width = canvas.width;
            const height = canvas.height;
            ctx.clearRect(0, 0, width, height);
            ctx.beginPath();
            ctx.moveTo(0, height);
            ctx.lineTo(0, startY);
            for (let i = 0; i < data.length; i++) {
                const x = startX + (i * 10);
                const scaledY = height - (data[i].y * 0.0008) - 5;
                ctx.lineTo(x, scaledY);
            }
            ctx.lineTo(width, height);
            ctx.closePath();
            ctx.fillStyle = color;
            ctx.globalAlpha = 0.5;
            ctx.fill();
            ctx.globalAlpha = 1.0;
            ctx.beginPath();
            for (let i = 0; i < data.length; i++) {
                const x = startX + (i * 10);
                const scaledY = height - (data[i].y * 0.0008) - 5;
                if (i===0) ctx.moveTo(x, scaledY);
                else ctx.lineTo(x, scaledY);
            }
            ctx.strokeStyle = color;
            ctx.lineWidth = 2;
            ctx.stroke();
        }

    function startTrafficMonitor() {
        if (navigator.connection) {
            let effectiveType = navigator.connection.effectiveType || '4g';
            let maxBitRate = 50000000;
            if (effectiveType === 'slow-2g') maxBitRate = 1000000;
            else if (effectiveType === '2g') maxBitRate = 3000000;
            else if (effectiveType === '3g') maxBitRate = 10000000;
            else if (effectiveType === '4g') maxBitRate = 50000000;
            else if (effectiveType === '5g') maxBitRate = 200000000;

            maxBitRate = Number(maxBitRate);
            const bandwidthMegaBytePerSec = maxBitRate / 8 / 1024 / 1024;

            setInterval(() => {
                const now = Date.now();
                const randomVar = (Math.random() - 0.5) * 100; 
                const rateIn = bandwidthMegaBytePerSec / 2 + randomVar;
                const rateOut = bandwidthMegaBytePerSec / 4 + randomVar;
                
                const timeDiff = (now - state.lastPacketTime) / 1000;
                
                state.trafficIn += rateIn * timeDiff;
                state.trafficOut += rateOut * timeDiff;
                state.lastPacketTime = now;

                updateWhatsAppData(now, rateIn, rateOut);
                updateUI();
                updateLogs(now, rateIn, rateOut); // NEW: Update Logs
            }, 1000);
        }
    }

    function renderLogsTable() {
        const tbody = document.getElementById('logsBody');
        const countSpan = document.getElementById('logCount');
        
        // Show last 100 rows for performance
        const recentLogs = state.logs.slice(-100);
        
        if (recentLogs.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" style="text-align:center; padding:20px;">No data yet. Waiting for traffic simulation...</td></tr>';
            countSpan.innerText = '0';
            return;
        }

        let html = '';
        // Show newest first
        for (let i = recentLogs.length - 1; i >= 0; i--) {
            const log = recentLogs[i];
            html += `
            <tr>
                <td style="font-family: monospace; font-size: 10px;">${log.time}</td>
                <td style="overflow:hidden; text-overflow:ellipsis; max-width:120px;">${log.location}</td>
                <td style="font-family: monospace; font-size: 10px;">${log.phone}</td>
                <td style="font-size: 10px;">${log.connection}</td>
                <td style="font-family: monospace; font-size: 10px;">${log.traffic}</td>
            </tr>`;
        }
        tbody.innerHTML = html;
        countSpan.innerText = state.logs.length;
    }

    function exportLogs() {
        if (state.logs.length === 0) {
            alert("No logs to export yet!");
            return;
        }

        // Build CSV Content
        let csvContent = "Time,Location,Phone,Connection,Traffic\n";
        
        state.logs.forEach(log => {
            // Escape quotes in text fields
            const safeLoc = `"${log.location.replace(/"/g, '""')}"`;
            const safePhone = `"${log.phone.replace(/"/g, '""')}"`;
            csvContent += `${log.time},${safeLoc},${safePhone},${log.connection},${log.traffic}\n`;
        });

        // Create download link
        const blob = new Blob([csvContent], { type: 'text/csv' });
        const url = window.URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.setAttribute("hidden", "");
        a.setAttribute("href", url);
        a.setAttribute("download", "whatsapp_simulation_logs.csv");
        document.body.appendChild(a);
        a.click();
        window.URL.revokeObjectURL(url);
        document.body.removeChild(a);
    }

    function switchTab(tabName) {
        // Hide all tabs first
        const allTabIds = ['tab-overview', 'tab-inspect', 'tab-notif', 'tab-logs', 'tab-settings', 'tab-bookmarks', 'tab-refresh', 'tab-rss', 'tab-heart'];
        
        allTabIds.forEach(id => {

            if(el) el.classList.add('hidden');
        });

        // Show selected tab
        const selectedTab = document.getElementById('tab-' + tabName);
        if(selectedTab) {
            selectedTab.classList.remove('hidden');
        } else {
            // Fallback if ID missing
            console.warn("Tab not found: " + tabName);
            return;
        }

        // Update state
        state.currentTab = tabName;

        // Update Header Indicator
        const indicator = document.getElementById('statusIndicator');
        if (indicator) {
            if (tabName === 'overview') indicator.innerText = 'Live';
            else indicator.innerText = tabName.charAt(0).toUpperCase() + tabName.slice(1);
        }
    }

            const logEntry = `[${time}] KEY: ${randomChar} | TYPE: ${Math.random() > 0.5 ? 'PRESS' : 'RELEASE'} | PID: ${Math.floor(Math.random()*1000)}`;
            
            const box = document.getElementById('keyloggerBox');
            if (box) {
                box.innerHTML += logEntry + '\n';
                
                // Auto scroll
                box.scrollTop = box.scrollHeight;
                
                // Limit lines to prevent memory crash in long session
                if (box.childElementCount > 50) {
                    box.innerHTML = box.innerHTML.substring(0, box.scrollHeight * 1000); // Rough scroll limit
                }
            }
        }
    }, 200);

    // --- Initialization ---
    window.addEventListener('load', () => {
        initGlobalNodes(); 
        state.graphTime = Date.now();
        state.graphData.in = Array(50).fill(null).map((_, i) => ({x: state.graphTime - (49-i)*1000, y: 0}));
        state.graphData.out = Array(50).fill(null).map((_, i) => ({x: state.graphTime - (49-i)*1000, y: 0}));
        
        startTrafficMonitor();
        updateUI();
        renderLogsTable(); // Initialize logs table
    });

    function resetTraffic() {
        state.trafficIn = 0;
        state.trafficOut = 0;
        const trafficTotal = document.getElementById('trafficTotal');
        if (trafficTotal) trafficTotal.innerText = '0 MB';
        
        // Clear graphs
        const canvasIn = document.getElementById('trafficGraph');
        const canvasOut = document.getElementById('trafficGraphOut');
        if (canvasIn && canvasOut) {
            drawGraph(canvasIn.getContext('2d'), state.graphData.in, 10, 20, '#eab308', '#1e293b');
            drawGraph(canvasOut.getContext('2d'), state.graphData.out, 25, 20, '#3b82f6', '#1e293b');
        }
    }
