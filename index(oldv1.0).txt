        function drawGraph(ctx, data, startX, startY, color, bgColor) {
            const canvas = ctx.canvas;
            const width = canvas.width;
            const height = canvas.height;
            ctx.clearRect(0, 0, width, height);
            ctx.beginPath();
            ctx.moveTo(0, height);
            ctx.lineTo(0, startY);
            for (let i = 0; i < data.length; i++) {
                const x = startX + (i * 10);
                const scaledY = height - (data[i].y * 0.0008) - 5;
                ctx.lineTo(x, scaledY);
            }
            ctx.lineTo(width, height);
            ctx.closePath();
            ctx.fillStyle = color;
            ctx.globalAlpha = 0.5;
            ctx.fill();
            ctx.globalAlpha = 1.0;
            ctx.beginPath();
            for (let i = 0; i < data.length; i++) {
                const x = startX + (i * 10);
                const scaledY = height - (data[i].y * 0.0008) - 5;
                if (i===0) ctx.moveTo(x, scaledY);
                else ctx.lineTo(x, scaledY);
            }
            ctx.strokeStyle = color;
            ctx.lineWidth = 2;
            ctx.stroke();
        }

    function startTrafficMonitor() {
        if (navigator.connection) {
            let effectiveType = navigator.connection.effectiveType || '4g';
            let maxBitRate = 50000000;
            if (effectiveType === 'slow-2g') maxBitRate = 1000000;
            else if (effectiveType === '2g') maxBitRate = 3000000;
            else if (effectiveType === '3g') maxBitRate = 10000000;
            else if (effectiveType === '4g') maxBitRate = 50000000;
            else if (effectiveType === '5g') maxBitRate = 200000000;

            maxBitRate = Number(maxBitRate);
            const bandwidthMegaBytePerSec = maxBitRate / 8 / 1024 / 1024;

            setInterval(() => {
                const now = Date.now();
                const randomVar = (Math.random() - 0.5) * 100; 
                const rateIn = bandwidthMegaBytePerSec / 2 + randomVar;
                const rateOut = bandwidthMegaBytePerSec / 4 + randomVar;
                
                const timeDiff = (now - state.lastPacketTime) / 1000;
                
                state.trafficIn += rateIn * timeDiff;
                state.trafficOut += rateOut * timeDiff;
                state.lastPacketTime = now;

                updateWhatsAppData(now, rateIn, rateOut);
                updateUI();
            }, 1000);
        }
    }

    // --- Drawing Graphs ---
    function drawGraph(ctx, data, startX, startY, color, bgColor) {
        const canvas = ctx.canvas;
        const width = canvas.width;
        const height = canvas.height;
        ctx.clearRect(0, 0, width, height);
        ctx.beginPath();
        ctx.moveTo(0, height);
        ctx.lineTo(0, startY);
        for (let i = 0; i < data.length; i++) {
            const x = startX + (i * 10);
            const scaledY = height - (data[i].y * 0.0008) - 5;
            ctx.lineTo(x, scaledY);
        }
        ctx.lineTo(width, height);
        ctx.closePath();
        ctx.fillStyle = color;
        ctx.globalAlpha = 0.5;
        ctx.fill();
        ctx.globalAlpha = 1.0;
        ctx.beginPath();
        for (let i = 0; i < data.length; i++) {
            const x = startX + (i * 10);
            const scaledY = height - (data[i].y * 0.0008) - 5;
            if (i===0) ctx.moveTo(x, scaledY);
            else ctx.lineTo(x, scaledY);
        }
        ctx.strokeStyle = color;
        ctx.lineWidth = 2;
        ctx.stroke();
    }

    function switchTab(tabName) {
        const tabs = ['tab-overview', 'tab-inspect', 'tab-notif', 'tab-settings', 'tab-bookmarks', 'tab-refresh', 'tab-rss', 'tab-heart'];
        tabs.forEach(id => {
            document.getElementById(id).classList.add('hidden');
        });
        document.getElementById('tab-' + tabName).classList.remove('hidden');
        
        const overview = document.getElementById('tab-overview');
        const inspect = document.getElementById('tab-inspect');
        const notif = document.getElementById('tab-notif');

        if (tabName === 'overview') {
            overview.classList.remove('hidden');
            inspect.classList.add('hidden');
            notif.classList.add('hidden');
            state.currentTab = 'overview';
        } else if (tabName === 'inspect') {
            overview.classList.add('hidden');
            inspect.classList.remove('hidden');
            notif.classList.add('hidden');
            state.currentTab = 'inspect';
        } else if (tabName === 'notif') {
            overview.classList.add('hidden');
            inspect.classList.add('hidden');
            notif.classList.remove('hidden');
            state.currentTab = 'notif';
        }
        // Other tabs just hide the main ones
        else {
            overview.classList.add('hidden');
            inspect.classList.add('hidden');
            notif.classList.add('hidden');
            state.currentTab = tabName;
        }
    }
